// ОТРИСОВКА 8 КАРТОЧЕК ЛИНИЙ С КОНТРОЛЕМ ТЕКУЩЕЙ СМЕНЫ function renderLiveLineCards() { const grid = document.getElementById('linesGrid'); grid.innerHTML = ''; const { start: shiftStart, end: shiftEnd } = getCurrentShiftWindow(); const shiftStartTime = shiftStart.getTime(); const shiftEndTime = shiftEnd.getTime(); PRIMATEK_LINES.forEach(lineNum => { // Ищем самую свежую запись вообще const latestAny = rawReports.find(r => Number(r.line_number) === lineNum); // Проверяем: относится ли этот замер к ТЕКУЩЕЙ активной смене? let isCurrentShift = false; if (latestAny && latestAny.created_at) { const recordTime = new Date(latestAny.created_at).getTime(); isCurrentShift = (recordTime >= shiftStartTime && recordTime <= shiftEndTime); } let statusClass = 'st-idle'; let statusText = 'НЕТ ДАННЫХ'; let op = '—'; let batch = '—'; let ral = '—'; let currentKg = 0; let chipsKg = 0; let boxText = '—'; let timeText = 'Замеров не было'; let centralOutputHtml = ''; if (latestAny && isCurrentShift) { // ДАННЫЕ ТЕКУЩЕЙ СМЕНЫ — ЛИНИЯ АКТИВНА statusText = latestAny.line_status || 'В РАБОТЕ'; if (statusText === 'В РАБОТЕ') statusClass = 'st-work'; else if (statusText === 'ЧИСТКА') statusClass = 'st-clean'; else if (statusText === 'ПРОБЫ') statusClass = 'st-sample'; else if (statusText === 'РЕМОНТ') statusClass = 'st-repair'; op = latestAny.operator_name || 'Не указан'; batch = latestAny.batch_number || '—'; ral = latestAny.ral_color || '—'; currentKg = Number(latestAny.total_good_kg) || 0; chipsKg = Number(latestAny.chips_kg) || 0; const bInfo = getBoxRange(latestAny); boxText = bInfo.text; const t = new Date(latestAny.created_at); timeText = `Замер в ${t.toLocaleTimeString('ru-RU', {hour:'2-digit', minute:'2-digit'})}`; if (statusText === 'В РАБОТЕ') { const chipsHtml = (chipsKg > 0) ? `
+ ${chipsKg} кг чипсы / п-ф
` : ''; centralOutputHtml = `
Текущий результат смены
${currentKg.toLocaleString('ru-RU')} кг
${boxText}
${chipsHtml}
${timeText}
`; } else { const chipsHtml = (chipsKg > 0) ? `
Чипсы/П-ф: ${chipsKg} кг
` : ''; centralOutputHtml = `
Состояние линии
${statusText}
Партия: ${batch}
${chipsHtml}
${timeText}
`; } } else if (latestAny && !isCurrentShift) { // ЛИНИЯ В ЭТУ СМЕНУ ЕЩЕ НЕ ПРИНЯТА ОПЕРАТОРОМ statusClass = 'st-idle'; statusText = 'НЕ ПРИНЯТА'; op = `Ожидает оператора`; batch = latestAny.batch_number || '—'; ral = latestAny.ral_color || '—'; const prevTime = new Date(latestAny.created_at); const prevTimeStr = prevTime.toLocaleDateString('ru-RU', {day:'2-digit', month:'2-digit'}) + ' ' + prevTime.toLocaleTimeString('ru-RU', {hour:'2-digit', minute:'2-digit'}); centralOutputHtml = `
Смена не начата
ОЖИДАНИЕ ПРИЁМКИ
Сдал: ${latestAny.operator_name || '—'}
${prevTimeStr}
`; } else { // В базе вообще никогда не было данных centralOutputHtml = `
Состояние линии
НЕТ ДАННЫХ
Партия: —
Замеров не было
`; } const card = document.createElement('div'); card.className = 'line-card'; if (!isCurrentShift && latestAny) { card.style.borderColor = '#1e293b'; card.style.opacity = '0.75'; } card.innerHTML = `
Линия №${lineNum} ${statusText}
Оператор: ${op}
Партия: ${batch} | ${ral}
${centralOutputHtml}
`; grid.appendChild(card); }); }
